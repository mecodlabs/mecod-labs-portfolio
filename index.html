// Kotlin MethodChannel Dispatcher
class SensorBridge(flutterEngine: FlutterEngine) {
    private val CHANNEL = "com.mecodlabs.pulseflow/telemetry"
    init {
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL).setMethodCallHandler { call, result ->
            if (call.method == "getHardwareVitals") {
                result.success(fetchRealtimeVitals())
            }
        }
    }
}
